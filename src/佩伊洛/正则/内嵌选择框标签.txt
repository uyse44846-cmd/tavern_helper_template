    <script>
        /* ========== 获取消息内容 ========== */
        function getMessageData() {
            var chatMessages = getChatMessages(getCurrentMessageId());
            if (!chatMessages || chatMessages.length === 0) return null;
            return chatMessages[0].message;
        }

        /* ========== 解析选项数据 ========== */
        function parseChoices(messageText) {
            var match = messageText.match(/<sprite_choices>([\s\S]*?)<\/sprite_choices>/);
            if (!match || !match[1]) return [];

            var lines = match[1].trim().split('\n').filter(function(l) { return l.trim(); });
            var choices = [];

            lines.forEach(function(line) {
                var m = line.match(/(.+?)[：:]\s*(.+)/);
                if (!m) return;

                var title = m[1].trim();
                var content = m[2].trim();

                /* 去除包裹引号 */
                if ((content.charAt(0) === '"' && content.charAt(content.length - 1) === '"') ||
                    (content.charAt(0) === '\u300c' && content.charAt(content.length - 1) === '\u300d') ||
                    (content.charAt(0) === '`' && content.charAt(content.length - 1) === '`')) {
                    content = content.slice(1, -1);
                }

                choices.push({ title: title, content: content });
            });

            return choices;
        }

        /* ========== HTML转义 ========== */
        function escapeHtml(str) {
            var div = document.createElement('div');
            div.textContent = str || '';
            return div.innerHTML;
        }

        /* ========== 填入输入框 ========== */
        function fillInput(text) {
            try {
                var input = window.parent.document.querySelector('#send_textarea');
                if (input) {
                    input.value = text;
                    input.dispatchEvent(new Event('input', { bubbles: true }));
                }
            } catch (e) {
                console.error(e);
            }
        }

        /* ========== 渲染选择卡片 ========== */
        function renderChoices(choices) {
            var grid = document.createElement('div');
            grid.className = 'peyro-choices-grid';

            choices.forEach(function(c) {
                var item = document.createElement('div');
                item.className = 'peyro-choice-item';

                var titleSpan = document.createElement('span');
                titleSpan.className = 'peyro-choice-title';
                titleSpan.textContent = c.title || '';

                var contentSpan = document.createElement('span');
                contentSpan.className = 'peyro-choice-content';
                contentSpan.textContent = c.content || '';

                item.appendChild(titleSpan);
                item.appendChild(contentSpan);

                item.addEventListener('click', (function(text) {
                    return function() { fillInput(text); };
                })(c.content));

                grid.appendChild(item);
            });

            document.getElementById('content').innerHTML = '';
            document.getElementById('content').appendChild(grid);
        }

        /* ========== 主函数 ========== */
        function init() {
            try {
                var msg = getMessageData();
                if (!msg) {
                    document.getElementById('content').innerHTML = '';
                    return;
                }

                var choices = parseChoices(msg);
                if (choices.length > 0) {
                    renderChoices(choices);
                } else {
                    document.getElementById('content').innerHTML = '';
                }
            } catch (e) {
                console.error(e);
                document.getElementById('content').innerHTML = '';
            }
        }

        $(function() { init(); });
    </script>
